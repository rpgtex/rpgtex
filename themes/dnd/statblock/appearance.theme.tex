%%%% modifying lengths in statblock
%% The tcolorbox style below holds the variables, not their values, so the setters only need to set the variables:
%% the values are read as each statblock opens
%% guarded, since a theme can be loaded more than once
\skip_if_exist:NF \__dnd_statParSkip    { \skip_new:N \__dnd_statParSkip }
\dim_if_exist:NF  \__dnd_statParIndent  { \dim_new:N  \__dnd_statParIndent }
\skip_if_exist:NF \__dnd_statActionSkip { \skip_new:N \__dnd_statActionSkip }

\NewDocumentCommand\RpgStatParSkip    {m} { \skip_set:Nn \__dnd_statParSkip    {#1} }
\NewDocumentCommand\RpgStatParIndent  {m} { \dim_set:Nn  \__dnd_statParIndent  {#1} }
\NewDocumentCommand\RpgStatActionSkip {m} { \skip_set:Nn \__dnd_statActionSkip {#1} }

%% set the initial defaults
\skip_set:Nn \__dnd_statParSkip    {0pt}
\dim_set:Nn  \__dnd_statParIndent  {1em}
\skip_set:Nn \__dnd_statActionSkip {0.5em}

\tcbset{rpgstatbox/.append~style={before~upper={
\skip_set_eq:NN \parskip   \__dnd_statParSkip
\dim_set_eq:NN  \parindent \__dnd_statParIndent
}}}

%% Appearance modification
\cs_set_protected_nopar:Npn \__dnd_appearancePreprocessing:
{
	\bool_if:NF{\__rpg_ShowBackground}
	{
		\colorlet{StatblockBackground}{white}
	}

	%% the theme's Statblock... colors are the defaults; these changes are local to this statblock
	\tl_if_blank:VF \__dnd_statColor
	{
		\colorlet{StatblockRule}{\tl_use:N \__dnd_statColor}
		\colorlet{StatblockHeading}{StatblockRule!50!black}
		\colorlet{StatblockFrame}{StatblockRule!80!white}
	}
	\tl_if_blank:VF \__dnd_statFrameColor
	{
		\colorlet{StatblockFrame}{\tl_use:N \__dnd_statFrameColor}
	}
	\RpgSetThemeColor{StatblockRule!25!white}

	% change filigree color to
	\tl_clear:N\__dnd_statBoxOptions
	\bool_if:NTF{\__dnd_statFiligreeOutline}
	{
		\colorlet{FiligreeColor}{StatblockFrame}
		\tl_set:Ne\__dnd_statBoxOptions{rpgfiligree}
	}
	{
		\bool_if:NT{\__dnd_statFiligree}
		{
			\tl_set:Ne\__dnd_statBoxOptions{rpgfiligree}
		}
	}
}


%%Core stat environments

% Title and Italicized text appearing immediately afterwards
\DeclareDocumentCommand {\RpgStatTitle} {m}
{
	\vspace{3pt}
	\ignorespaces\noindent	{\RpgFontStatBlockTitle #1}

	\__rpg_join:NVVn{\__dnd_statSubtitle}{\__dnd_statSize}{\__dnd_statType}{\ }
	\__rpg_join:NVVn{\__dnd_statSubtitle}{\__dnd_statSubtitle}{\__dnd_statAlignment}{,\ }


	\tl_if_empty:eF{\__dnd_statSource}
	{
		\par \noindent\begin{minipage}[t]{0.5\linewidth}
			}
			\tl_if_empty:eF{\__dnd_statSubtitle}
			{
				\begin {hangingpar}
				\textit {\__dnd_statSubtitle}
				\end {hangingpar}
			}
			\tl_if_empty:eF{\__dnd_statSource}
			{
		\end{minipage}
		\begin{minipage}[t]{0.5\linewidth}
			\raggedleft \textit{Source:~\__dnd_statSource}
		\end{minipage}
	}
}

%%Formats Speed, hp, ability scores etc
\DeclareDocumentCommand {\RpgHeaderBlock} {m}
{
	%% the statblock's \parskip is for its body text: the header is spaced by hand
	\skip_zero:N \parskip
	\RpgStatTitle{#1}
	\RpgStatLine
	\__dnd_stat_basics:
	\RpgStatLine
	\__dnd_stat_ability_scores:
	\RpgStatLine
	\__dnd_stat_details:
	\RpgStatLine
	\skip_set_eq:NN \parskip \__dnd_statParSkip
	\vspace{4pt}
}

%% BASIC BLOCK -- HP / AC / etc.
\cs_set_protected_nopar:Npn \__dnd_stat_basics:
{
	{
			\color{StatblockHeading}
			\noindent
			\begin{tabularx}{\linewidth}{lXlX}
				\textbf{AC}         & \RaggedRight\noindent \__dnd_statAC
				                    &
				\textbf{Speed}      & \RaggedRight\noindent \__dnd_statSpeed
				\\
				\textbf{HP}         & \RaggedRight\noindent \exp_args:Ne\RpgDice{\__dnd_statHP}
				                    &
				\textbf{Initiative} & \RaggedRight\noindent \__dnd_statInitiativeBonus
			\end{tabularx}
		}
}
%% ABILITY SCORE BLOCK
%%Formatters for ability table

\cs_set:Npn \__dnd_format_mod_labels:n #1
{
	\rotatebox[origin=c]{90}{\RpgFontStatBlockTitle\scriptsize \hspace{0.1cm} #1 \hspace{0.1cm}}
}
\cs_set:Npn \__dnd_format_stat_titles:nn #1#2 {
	%% \leavevmode starts the paragraph before the font's \color: otherwise the colour lands in vertical mode at the top
	%% of the (p-type) cell, and the cell drops a line below the row.
	%% In a Y cell, \linewidth is the width of the column: a title too wide for its column is shrunk to fit, rather than wrapping
	&\leavevmode\RpgFontStatBlockTitle \normalsize\maxsizebox{\linewidth}{!}{#1\ (#2)}
}


\cs_set_protected_nopar:Npn \__dnd_stat_ability_scores:
{
	%%define colours and spacings
	\color {StatblockHeading}
	\setlength\topsep{0pt}
	\setlength\tabcolsep{3pt}
	\begin{center}
		\begin{RpgTable}[vskip=0pt,noheader]{rYYYYYY}
			~
			\seq_map_inline:Nn\__dnd_Stats
			{
				\__dnd_format_stat_titles:nn{\text_uppercase:n{##1}}{\use:c{__dnd_##1Score}}
			}
			\\
			\__dnd_format_mod_labels:n{mod}
			\seq_map_inline:Nn\__dnd_Stats
			{
				& \__dnd_format_signed_integer:N{\use:c{__dnd_##1Mod}}
			}

			%if any proficiencies set, then render the save row
			\__dnd_if_any_saves:T
			{
				\\
				\__dnd_format_mod_labels:n{save}
				\seq_map_inline:Nn{\__dnd_Stats}
				{
					& \tl_use:c{__##1Save}
				}
			}
		\end{RpgTable}
	\end{center}
	\par \vspace{0.1cm}
}

%% DETAILS BLOCK -- languages/skilsl etc
\cs_set_protected_nopar:Npn \__dnd_stat_details:
{
	%% Each item is wrapped in command that ensures it only appears 
	%% if the value is non-empty
	\begin {description}[
		before   = \color {StatblockHeading},
		font     = \RpgFontStatBlockBody\color {StatblockHeading},
		labelsep = \__rpg_SpaceDim,
		nosep]

	\__maybe_item:ne {Skills}					{\__dnd_statProcessedSkills} %% use e variant to force expansion if empty
	\__maybe_item:nn {Damage~Vulnerabilities}	{\__dnd_statDamageWeak}
	\__maybe_item:nn {Damage~Resistances}		{\__dnd_statDamageResist}
	\__maybe_item:nn {Damage~Immunities}		{\__dnd_statDamageImmune}
	\__maybe_item:nn {Condition~Immunities}		{\__dnd_statConditionImmune}
	\__maybe_item:nn {Senses}					{\__dnd_construct_perception:}
	\__maybe_item:nn {Languages}				{\__dnd_statLanguages}
	\__maybe_item:nn {Challenge}				{\__dnd_statCR{}~(\RpgNumber{\__dnd_statXP}xp,~ proficiency:~\__dnd_format_signed_integer:N {\__dnd_statPB})}
	\end {description}
}
