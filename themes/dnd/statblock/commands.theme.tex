% Creates a section within environment
% aliased by 'section' within the environment
\DeclareDocumentCommand {\RpgStatSection} {m}
{
	{
			\par\addvspace{6pt plus 2pt minus 2pt}
			\vbox{
				\noindent\RpgFontStatBlockSection #1
				\vspace {2pt}
				\hrule height 0.6pt
				\par
			}}\nopagebreak\ignorespaces\noindent
}

% Inline header for stat actions - similar to a paragraph
% aliased in the environment by \action
% optional arg is used by legendary actions

\bool_set_false:N \__dnd_inLegendary
\bool_if_exist:NF \__dnd_statInSubActions { \bool_new:N \__dnd_statInSubActions }
\DeclareDocumentCommand {\RpgStatAction} {O{} m}
{
	\par
	%% no skip at the top of the box (\prevdepth is -1000pt), or straight after a \RpgStatSection (which ends with
	%% \nopagebreak, so the last item on the list is a 10000 penalty)
	\bool_lazy_or:nnF
	{ \dim_compare_p:nNn { \prevdepth } < { -999pt } }
	{ \int_compare_p:nNn { \lastpenalty } = { 10000 } }
	{ \skip_vertical:N \__dnd_statActionSkip }
	\noindent {
		\bool_if:NTF{\__dnd_statInSubActions}{\gillius\bfseries}{\RpgFontParagraph }#2

		\bool_if:NTF\__dnd_inLegendary
		{
			{\tl_if_blank:eF{#1}{~({\RpgPlural{#1}{action}})}}
		}
		{
		}
		.
	}
}

% An action whose body is a run of sub-actions (the attacks of a multiattack, a set of options, a roll table), indented
% under a bar. #1 = the action's name, #2 = the text on its line, before the sub-actions
% aliased in the environment by subactions, and by multiattack (with the name defaulting to Multiattack)
\tcbset{rpgstatsubactions/.style={
	blank, breakable,
	borderline~west={0.8pt}{0pt}{StatblockRule},
	left=10pt, before~skip=2pt, after~skip=5pt,
}}
\DeclareDocumentEnvironment{RpgStatSubActions}{m m}
{
	\RpgStatAction{#1}
	\tl_if_blank:nF{#2}{~#2}
	\par
	%% local to the environment, so a nested block restores its parent's setting when it ends
	\bool_set_true:N \__dnd_statInSubActions
	\begin{tcolorbox}[rpgstatsubactions]
}
{
	\end{tcolorbox}
}

% As with action, but allows for an italicised trigger
\DeclareDocumentCommand\RpgStatReaction{m m}
{
	\RpgStatAction{#1}
	\tl_if_blank:eF{#2}{~\textit{#2:}}
}

% Interface for internal values

\DeclareDocumentCommand\RpgStatSetName{m}
{
	\tl_set:Nn \__dnd_statName {#1}
}

\DeclareDocumentCommand\RpgStatName{s s}
{
	\IfBooleanT{#1}{\IfBooleanTF{#2}{\MakeTitlecase{\__dnd_statArticle}}{\__dnd_statArticle}}
	\__dnd_statName{}
}
\DeclareDocumentCommand{\RpgStatMod}{m}{
	\__dnd_format_signed_integer:N{\str_case:nn{#1}
		{
			{proficiency}{\__dnd_statPB}
				{str}{\__dnd_strMod}
				{dex}{\__dnd_dexMod}
				{con}{\__dnd_conMod}
				{wis}{\__dnd_wisMod}
				{int}{\__dnd_intMod}
				{cha}{\__dnd_chaMod}
		}}
}
